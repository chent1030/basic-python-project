## Agent 间通讯（3 种模式）

### 1. 共享黑板（Blackboard）—— 所有 agent 读写同一空间

```python
class Blackboard:
    """共享黑板。任一 agent 写入的数据,其它 agent 能读到。
    解耦:写入方不关心谁读,读取方不关心谁写。"""
    _data: dict[str, BlackboardEntry]    # key -> {value, writer, timestamp}

    def write(self, key: str, value: Any, writer: str = ""): ...
    def read(self, key: str) -> Any | None: ...
    def has(self, key: str) -> bool: ...
    def keys(self) -> list[str]: ...
    def history(self, key: str) -> list[BlackboardEntry]: ...  # 写入历史
```

业务用法：
```python
class Researcher(BaseSingleAgent):
    async def run(self, message):
        result = await self._invoke_backend(...)
        self.write("research", result)     # 写到黑板
        return result

class Writer(BaseSingleAgent):
    async def run(self, message):
        research = self.read("research")   # 读 Researcher 写的
        ...
```

### 2. 消息传递（MessageBus）—— agent 间直接互发

```python
class MessageBus:
    """消息传递。agent A 直接给 agent B 发消息,B 处理后可回复。
    精确控制发给谁,支持请求-响应和通知两种模式。"""
    _agents: dict[str, BaseAgent]          # name -> 实例(注册后可互发)
    _mailbox: dict[str, list[Message]]     # name -> 收件箱

    def register(self, agent: BaseAgent): ...
    async def send(self, target: str, content: str, sender: str = ""): ...
    async def request(self, target: str, content: str, sender: str = "") -> str: ...  # 等回复
    def receive(self, name: str) -> list[Message]: ...
    def clear_mailbox(self, name: str): ...
```

业务用法：
```python
class Critic(BaseSingleAgent):
    async def run(self, message):
        result = await self._invoke_backend(...)
        # 直接给 Writer 发修改建议
        await self.send("writer", f"建议修改:{result}")
        return result

class Writer(BaseSingleAgent):
    async def run(self, message):
        msgs = self.receive_messages()      # 收件箱
        for m in msgs:
            ...  # 处理 Critic 的建议
```

### 3. 事件总线（EventBus）—— 发布/订阅,松耦合触发

```python
class EventBus:
    """事件总线。agent 发布事件,订阅者自动被调用。
    松耦合:发布者不关心谁订阅,适合「X完成后触发Y」。"""
    _subscribers: dict[str, list[EventHandler]]

    def subscribe(self, event_type: str, handler: EventHandler): ...
    async def publish(self, event_type: str, data: Any, source: str = ""): ...
    def unsubscribe(self, event_type: str, handler: EventHandler): ...
```

业务用法：
```python
class OcrAgent(BaseSingleAgent):
    async def run(self, message):
        result = await self._invoke_backend(...)
        await self.publish("ocr_done", result)    # 发布事件
        return result

class Notifier(BaseSingleAgent):
    def setup(self):
        self.subscribe("ocr_done", self.on_ocr)   # 订阅

    async def on_ocr(self, event):
        # OCR 完成后自动触发,不需要 OcrAgent 知道 Notifier 存在
        ...
```

### BaseAgent 提供的通讯 API（业务直接调）

```python
class BaseAgent:
    # 框架注入(业务代码直接用,不用管怎么创建)
    # blackboard / message_bus / event_bus 由 Pipeline/Gateway 在 run() 时注入

    # ---- 黑板 ----
    def write(self, key: str, value: Any): ...      # self.blackboard.write(key, value, self.name)
    def read(self, key: str) -> Any | None: ...      # self.blackboard.read(key)

    # ---- 消息 ----
    async def send(self, target: str, content: str): ...
    async def request(self, target: str, content: str) -> str: ...  # 等回复
    def receive_messages(self) -> list[Message]: ...

    # ---- 事件 ----
    async def publish(self, event_type: str, data: Any): ...
    def subscribe(self, event_type: str, handler): ...
```

### 通讯基础设施的生命周期

```python
class AgentRunContext:
    """每次 Pipeline 运行创建一套通讯基础设施,所有 agent 共享。"""
    blackboard: Blackboard       # 共享黑板(同一 Pipeline 内所有 agent 共享)
    message_bus: MessageBus      # 消息总线(注册了所有成员 agent)
    event_bus: EventBus          # 事件总线
```

Pipeline/Gateway 在 run() 开头创建 Context(含三种通讯设施),传给每个 agent。
复合拓扑(parallel/pipeline/conversational)的成员 agent 自动注册到 message_bus。

## BaseAgent 完整设计（汇总全部能力）

```python
class BaseAgent:
    # ---- 声明式配置 ----
    name: str = ""
    backend: str = "deepagents"          # deepagents | agentscope | llm
    provider: str = ""
    system_prompt: str = ""
    tools: list[str] = []
    middleware: list[str] = []
    model/temperature/max_tokens/recursion_limit/skills: ...
    hitl_require_confirmation: bool = False

    # ---- 框架注入(运行时)----
    context: AgentRunContext              # 含 blackboard/message_bus/event_bus/messages

    # ---- 框架提供的执行方法 ----
    async def run(self, message, *, session_id, source) -> AgentResult: ...   # 主流程
    async def stream(self, message, ...) -> AsyncIterator[str]: ...           # 流式
    async def _invoke_backend(self, messages) -> str: ...                    # 调后端

    # ---- 通讯 API ----
    def write/read(self, ...): ...        # 黑板
    async def send/request/receive_messages(self, ...): ...  # 消息
    async def publish/subscribe(self, ...): ...              # 事件

    # ---- 中间件 ----
    async def _run_middleware_before/after(self, ...): ...   # 洋葱模型
```
